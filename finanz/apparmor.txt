#include <tunables/global>

  /usr/sbin/** rix,

  /usr/share/** r,
  /etc/** r,
  /proc/** r,
  /sys/fs/cgroup/** r,
  /dev/null rw,
  /dev/urandom r,
  /dev/tty rw,

  # App-Code (nur lesen) und Python-Cache
  /app/** r,
