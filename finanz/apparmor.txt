#include <tunables/global>

# AppArmor-Profil der Finanzplanungs-App (Name = slug), nach der Vorlage der Home-Assistant-Entwicklerdoku.
# Dateizugriff ist wie im Container erlaubt; eingeschränkt werden Fähigkeiten (nur Docker-Standard),
# mount, ptrace und Zugriffe auf Kernel-Schnittstellen.
profile finanz flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Fähigkeiten: höchstens der Docker-Standardsatz
  capability chown,
  capability dac_override,
  capability dac_read_search,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability setpcap,
  capability net_bind_service,

  # Netzwerk: Webserver (Ingress) und Aufrufe an den Supervisor (KI-Aufgabe)
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,

  signal,

  # Dateien
  file,

  # Programme
  /run.sh rix,
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/local/bin/** rix,
  /init ix,
  /command/** ix,
  /package/** ix,
  /run/{s6,s6-rc*,service}/** ix,

  # Daten der App
  /data/** rw,
  /config/** rwk,
  /app/** r,
  /tmp/** rwk,
  /run/{,**} rwk,

  # Kein Zugriff auf Kernel-Schnittstellen
  deny /sys/firmware/** rwklx,
  deny /sys/kernel/security/** rwklx,
  deny /proc/sys/kernel/** wklx,
  deny /proc/kcore rwklx,
  deny mount,
  deny umount,
  deny ptrace,
}
